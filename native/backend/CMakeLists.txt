cmake_minimum_required(VERSION 3.23)
project(cie_portable_backend LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
add_subdirectory(cie_sign_sdk)
include(cmake/MiddlewareSources.cmake)
list(FILTER CIE_MIDDLEWARE_SOURCES EXCLUDE REGEX "/(CacheLib|Logger)\\.cpp$|/UI/AbilitaCIE\\.mm$")
add_library(cie-pkcs11 SHARED ${CIE_MIDDLEWARE_SOURCES} platform/Cache.cpp platform/Logger.cpp platform/Ui.cpp)
target_include_directories(cie-pkcs11 PRIVATE cie-pkcs11/Util cie-pkcs11/PKCS11 cie-pkcs11/Sign platform)
target_link_libraries(cie-pkcs11 PRIVATE cie_sign_sdk)
target_compile_definitions(cie-pkcs11 PRIVATE CRYPTOKI_EXPORTS)
set_target_properties(cie-pkcs11 PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MINGW)
  target_link_options(cie-pkcs11 PRIVATE -Wl,--export-all-symbols)
endif()
if(WIN32)
  set_target_properties(cie-pkcs11 PROPERTIES PREFIX "")
endif()
if(APPLE)
  set_target_properties(cie-pkcs11 PROPERTIES INSTALL_NAME_DIR "@rpath")
endif()
if(CIE_BUILD_TESTS AND NOT WIN32)
  add_executable(cie_middleware_tests tests/middleware_tests.cpp)
  target_include_directories(cie_middleware_tests PRIVATE cie-pkcs11/PKCS11 platform)
  target_link_libraries(cie_middleware_tests PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME cie_middleware_tests COMMAND cie_middleware_tests "$<TARGET_FILE:cie-pkcs11>")
  add_executable(cie_verification_tests tests/verification_bridge_tests.cpp)
  target_include_directories(cie_verification_tests PRIVATE cie-pkcs11/Sign cie-pkcs11/Util platform)
  target_link_libraries(cie_verification_tests PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME cie_verification_tests COMMAND cie_verification_tests "${CMAKE_CURRENT_BINARY_DIR}/cie_sign_sdk/verification-fixtures" "$<TARGET_FILE:cie-pkcs11>")
  if(NOT APPLE)
    add_test(NAME cie_middleware_glib_load COMMAND cie_middleware_tests "$<TARGET_FILE:cie-pkcs11>" --preload-glib)
    target_link_libraries(cie_middleware_tests PRIVATE PkgConfig::PCSC)
    target_link_libraries(cie_verification_tests PRIVATE PkgConfig::PCSC)
  endif()
  set_tests_properties(cie_verification_tests PROPERTIES FIXTURES_REQUIRED CIEVerificationFixtures)
endif()
install(TARGETS cie-pkcs11 RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(DIRECTORY LICENSES DESTINATION licenses/backend)
install(FILES LICENSE.txt THIRD_PARTY_NOTICES.md LICENSE_AUDIT.md DESTINATION licenses/backend)
if(CIE_BUILD_TESTS)
  add_executable(cie_handle_tests tests/handle_tests.cpp)
  target_include_directories(cie_handle_tests PRIVATE cie-pkcs11/PKCS11 cie-pkcs11/Util platform)
  target_link_libraries(cie_handle_tests PRIVATE cie-pkcs11 cie_sign_sdk)
  if(APPLE)
    set_target_properties(cie_handle_tests PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "@loader_path;@loader_path/..")
  elseif(UNIX)
    set_target_properties(cie_handle_tests PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN;$ORIGIN/..")
  endif()
  add_test(NAME cie_handle_tests COMMAND cie_handle_tests)
endif()
