# This directory is built only as part of the Linux native runtime.
find_package(Threads REQUIRED)
set_source_files_properties(pkcs11.cpp card.cpp snap_public_store.cpp tests/fake_card.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
set(CIE_LINUX_WEB_SOURCES pkcs11.cpp card.cpp snap_public_store.cpp ../../../src/local_cache.cpp ../../../src/local_cache_crypto.cpp)
function(cie_linux_web_target target)
  target_include_directories(${target} PRIVATE ../../../include ../../../src)
  target_link_libraries(${target} PRIVATE OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
  set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES BUILD_WITH_INSTALL_RPATH ON
    INSTALL_RPATH "$ORIGIN")
  # The legacy middleware exports the same PKCS#11 names. Keep this provider's
  # function table bound to its own entry point, independent of load order.
  target_link_options(${target} PRIVATE "LINKER:-Bsymbolic-functions" "LINKER:--exclude-libs,ALL")
endfunction()
add_library(cie_linux_pkcs11 SHARED ${CIE_LINUX_WEB_SOURCES})
cie_linux_web_target(cie_linux_pkcs11)
target_link_libraries(cie_linux_pkcs11 PRIVATE cie_web PkgConfig::PCSC)
set_target_properties(cie_linux_pkcs11 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
install(TARGETS cie_linux_pkcs11 LIBRARY DESTINATION .)
