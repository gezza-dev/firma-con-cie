cmake_minimum_required(VERSION 3.23)
project(cie_prototype_minidriver LANGUAGES CXX)
if(MSVC)
  add_compile_options(/utf-8)
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The prototype minidriver currently requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
# CPDK v7 header. Older MinGW installations ship a v4-only cardmod.h, which
# must not silently compile a different ABI.
# Never use the modified legacy CardMod.h from cie_sign_sdk/include.
set(CIE_CARDMOD_INCLUDE "" CACHE PATH "Directory containing CPDK cardmod.h")
add_library(cie_prototype_minidriver SHARED minidriver.cpp ../../../src/paired_store.cpp ../../../src/pairing_invalidation.cpp ../../../src/web_cert_store.cpp ../../../src/local_cache.cpp ../../../src/local_cache_crypto.cpp)
target_include_directories(cie_prototype_minidriver BEFORE PRIVATE "${CIE_CARDMOD_INCLUDE}")
target_include_directories(cie_prototype_minidriver PRIVATE ../../../include ../../../src)
target_compile_definitions(cie_prototype_minidriver PRIVATE CIE_WEB_NOEXPORT NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(cie_prototype_minidriver PRIVATE winscard crypt32 advapi32 bcrypt ncrypt)
set_target_properties(cie_prototype_minidriver PROPERTIES PREFIX "")
add_executable(cie_web_registration registration.cpp)
target_compile_definitions(cie_web_registration PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(cie_web_registration PRIVATE winscard advapi32 shell32 ole32 uuid)
if(MSVC)
 target_sources(cie_prototype_minidriver PRIVATE minidriver.def)
 set_property(TARGET cie_prototype_minidriver PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(MINGW)
 target_link_options(cie_prototype_minidriver PRIVATE -static -static-libgcc -static-libstdc++)
 target_link_options(cie_web_registration PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
set(CIE_WEB_SETUP_MANIFEST "" CACHE FILEPATH "Generated hash-pinned setup_payload.h")
if(CIE_WEB_SETUP_MANIFEST)
 get_filename_component(CIE_WEB_SETUP_INCLUDE "${CIE_WEB_SETUP_MANIFEST}" DIRECTORY)
 add_executable(cie_web_setup setup.cpp)
 target_include_directories(cie_web_setup PRIVATE "${CIE_WEB_SETUP_INCLUDE}" ../../../include ../../../src)
 target_compile_definitions(cie_web_setup PRIVATE CIE_WEB_NOEXPORT NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
 target_link_libraries(cie_web_setup PRIVATE winscard crypt32 bcrypt advapi32 shell32 ole32 uuid)
 set_target_properties(cie_web_setup PROPERTIES OUTPUT_NAME "CIE-Web-Setup")
 if(MSVC)
  set_property(TARGET cie_web_setup PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 endif()
 if(MINGW)
  target_link_options(cie_web_setup PRIVATE -municode -static -static-libgcc -static-libstdc++)
 endif()
endif()
if(BUILD_TESTING)
 add_executable(cie_web_catalog_tests setup_catalog_tests.cpp)
 target_link_libraries(cie_web_catalog_tests PRIVATE advapi32)
 add_test(NAME web_card_catalog COMMAND cie_web_catalog_tests)
 add_executable(cie_installer_fake installer_fake.cpp)
 if(MSVC)
  set_property(TARGET cie_installer_fake PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 endif()
 add_executable(cie_web_setup_policy_tests setup_policy_tests.cpp)
 target_include_directories(cie_web_setup_policy_tests PRIVATE ../../../src)
 target_link_libraries(cie_web_setup_policy_tests PRIVATE shell32)
 add_test(NAME web_setup_policy COMMAND cie_web_setup_policy_tests)
 add_library(cie_web_fake SHARED ../../../tests/fake_web.cpp)
 target_include_directories(cie_web_fake PRIVATE ../../../include ../../../src)
 set_target_properties(cie_web_fake PROPERTIES PREFIX "" OUTPUT_NAME cie_web
   RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tests/provider/backend")
 add_executable(cie_minidriver_tests minidriver_tests.cpp)
 target_include_directories(cie_minidriver_tests BEFORE PRIVATE "${CIE_CARDMOD_INCLUDE}")
 target_include_directories(cie_minidriver_tests PRIVATE ../../../src)
 target_compile_definitions(cie_minidriver_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
 add_custom_command(TARGET cie_prototype_minidriver POST_BUILD
   COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/tests/provider"
   COMMAND "${CMAKE_COMMAND}" -E copy "$<TARGET_FILE:cie_prototype_minidriver>" "${CMAKE_CURRENT_BINARY_DIR}/tests/provider/")
 add_dependencies(cie_minidriver_tests cie_prototype_minidriver cie_web_fake)
 add_test(NAME isolated_minidriver COMMAND cie_minidriver_tests "${CMAKE_CURRENT_BINARY_DIR}/tests/provider/cie_prototype_minidriver.dll")
 if(MINGW)
  target_link_options(cie_web_fake PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(cie_minidriver_tests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
endif()
