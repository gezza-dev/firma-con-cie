cmake_minimum_required(VERSION 3.23)
project(cie_windows_pkcs11 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CIE_OPENSSL_INCLUDE "" CACHE PATH "Headers matching the private backend's OpenSSL")
set(CIE_BACKEND_IMPORT "" CACHE FILEPATH "Import library for the unchanged private backend")
set(CIE_WEB_IMPORT "" CACHE FILEPATH "Import library for the unchanged Web core")
set(CIE_BACKEND_RUNTIME "" CACHE PATH "Hash-validated private runtime")
set(WEB ../../linux/web)
set(CACHE_SOURCES ../../../src/local_cache.cpp ../../../src/local_cache_crypto.cpp)
function(configure_provider target)
  target_include_directories(${target} PRIVATE ${CIE_OPENSSL_INCLUDE} ../../../include ../../../src)
  target_compile_definitions(${target} PRIVATE CIE_PKCS11_TABLE_ONLY CIE_WEB_NOEXPORT NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(${target} PRIVATE "${CIE_BACKEND_IMPORT}" bcrypt advapi32 crypt32)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endfunction()
add_library(cie_windows_pkcs11 SHARED ${WEB}/pkcs11.cpp ${WEB}/card.cpp ${CACHE_SOURCES})
target_sources(cie_windows_pkcs11 PRIVATE pkcs11.def)
configure_provider(cie_windows_pkcs11)
target_link_libraries(cie_windows_pkcs11 PRIVATE "${CIE_WEB_IMPORT}" winscard)
set_target_properties(cie_windows_pkcs11 PROPERTIES PREFIX "" OUTPUT_NAME "cie_pkcs11")
foreach(name cie-pkcs11.dll cie_web.dll libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
  add_custom_command(TARGET cie_windows_pkcs11 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CIE_BACKEND_RUNTIME}/${name}" "$<TARGET_FILE_DIR:cie_windows_pkcs11>/${name}")
endforeach()
if(BUILD_TESTING)
  add_library(cie_windows_pkcs11_fake SHARED ${WEB}/pkcs11.cpp ${WEB}/card.cpp ${WEB}/tests/fake_card.cpp ${CACHE_SOURCES})
  target_sources(cie_windows_pkcs11_fake PRIVATE pkcs11.def)
  configure_provider(cie_windows_pkcs11_fake)
  # All PC/SC and Web engine functions are local test doubles. No winscard link.
  target_compile_definitions(cie_windows_pkcs11_fake PRIVATE CIE_PAIRING_TEST_SERVICE _WINSCARD_)
  add_executable(cie_windows_pkcs11_tests ${WEB}/tests/pkcs11_tests.cpp)
  configure_provider(cie_windows_pkcs11_tests)
  add_test(NAME windows_pkcs11_isolated COMMAND cie_windows_pkcs11_tests "$<TARGET_FILE:cie_windows_pkcs11_fake>")
endif()
