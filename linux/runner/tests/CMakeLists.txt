cmake_minimum_required(VERSION 3.13)
project(cie_linux_runner_tests LANGUAGES CXX)
enable_testing()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
get_filename_component(FLUTTER_LINUX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../flutter/ephemeral" ABSOLUTE)
add_executable(text_scaling_tests text_scaling_test.cc ../text_scaling.cc)
target_compile_features(text_scaling_tests PRIVATE cxx_std_14)
target_compile_options(text_scaling_tests PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
target_include_directories(text_scaling_tests PRIVATE "${FLUTTER_LINUX_DIR}" ..)
target_link_libraries(text_scaling_tests PRIVATE PkgConfig::GTK "${FLUTTER_LINUX_DIR}/libflutter_linux_gtk.so")
add_test(NAME linux_text_scaling COMMAND text_scaling_tests)
