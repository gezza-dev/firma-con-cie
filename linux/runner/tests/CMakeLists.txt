cmake_minimum_required(VERSION 3.13)
project(cie_linux_runner_tests LANGUAGES CXX)
enable_testing()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.22)
get_filename_component(FLUTTER_LINUX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../flutter/ephemeral" ABSOLUTE)
add_executable(text_scaling_tests text_scaling_test.cc ../text_scaling.cc)
target_compile_features(text_scaling_tests PRIVATE cxx_std_14)
target_compile_options(text_scaling_tests PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
target_include_directories(text_scaling_tests PRIVATE "${FLUTTER_LINUX_DIR}" ..)
target_link_libraries(text_scaling_tests PRIVATE PkgConfig::GTK "${FLUTTER_LINUX_DIR}/libflutter_linux_gtk.so")
add_test(NAME linux_text_scaling COMMAND text_scaling_tests)
add_executable(window_sizing_tests window_sizing_test.cc ../window_sizing.cc)
target_compile_features(window_sizing_tests PRIVATE cxx_std_14)
target_compile_options(window_sizing_tests PRIVATE -Wall -Wextra -Werror)
target_include_directories(window_sizing_tests PRIVATE ..)
target_link_libraries(window_sizing_tests PRIVATE PkgConfig::GTK)
add_test(NAME linux_window_sizing COMMAND window_sizing_tests)

# This optional GTK test uses only a virtual display and a private session bus.
find_program(XVFB_RUN xvfb-run)
find_program(DBUS_RUN_SESSION dbus-run-session)
if(XVFB_RUN AND DBUS_RUN_SESSION)
  add_executable(window_chrome_tests window_chrome_test.cc ../window_chrome.cc)
  target_compile_features(window_chrome_tests PRIVATE cxx_std_14)
  target_compile_options(window_chrome_tests PRIVATE -Wall -Wextra -Werror)
  target_include_directories(window_chrome_tests PRIVATE ..)
  target_link_libraries(window_chrome_tests PRIVATE PkgConfig::GTK)
  add_test(NAME linux_window_chrome COMMAND "${DBUS_RUN_SESSION}" -- "${XVFB_RUN}" -a "$<TARGET_FILE:window_chrome_tests>")
endif()
